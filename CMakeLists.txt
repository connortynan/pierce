cmake_minimum_required(VERSION 3.13)
project(pierce VERSION 0.1.0 LANGUAGES C)

option(PIERCE_BUILD_TESTS "Build tests" ON)
option(PIERCE_BUILD_TOOLS "Build the reference relay" ON)
option(PIERCE_BUILD_EXAMPLES "Build examples" ON)
option(PIERCE_SANITIZE "Build with ASan + UBSan" OFF)
option(PIERCE_TSAN "Build with TSan" OFF)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    set(PIERCE_WARNINGS -Wall -Wextra -Wshadow -Wstrict-prototypes -Wmissing-prototypes
                        -Wno-unused-function)
    if(PIERCE_SANITIZE)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
        add_link_options(-fsanitize=address,undefined)
    endif()
    if(PIERCE_TSAN)
        add_compile_options(-fsanitize=thread)
        add_link_options(-fsanitize=thread)
    endif()
elseif(MSVC)
    set(PIERCE_WARNINGS /W4)
endif()

# Sources: everything in src/ (platform files compile to nothing elsewhere).
file(GLOB PIERCE_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.c)

# Monocypher is a git submodule, built as-is without our warning flags.
if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/vendor/monocypher/src/monocypher.c)
    message(FATAL_ERROR "vendor/monocypher is empty: run `git submodule update --init`")
endif()
add_library(pierce_monocypher OBJECT vendor/monocypher/src/monocypher.c)
set_target_properties(pierce_monocypher PROPERTIES POSITION_INDEPENDENT_CODE ON
                      C_VISIBILITY_PRESET hidden)

add_library(pierce ${PIERCE_SOURCES} $<TARGET_OBJECTS:pierce_monocypher>)
target_include_directories(pierce PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_compile_definitions(pierce PRIVATE PIERCE_BUILD)
target_compile_options(pierce PRIVATE ${PIERCE_WARNINGS})
set_target_properties(pierce PROPERTIES C_VISIBILITY_PRESET hidden POSITION_INDEPENDENT_CODE ON)
if(BUILD_SHARED_LIBS)
    target_compile_definitions(pierce PUBLIC PIERCE_SHARED)
endif()

if(WIN32)
    target_link_libraries(pierce PUBLIC ws2_32 iphlpapi bcrypt)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(pierce PUBLIC Threads::Threads)
endif()

# Internal objects exposed for tests and the relay tool (they use internal
# headers and symbols that a hidden-visibility shared library would not
# export).
add_library(pierce_internal STATIC ${PIERCE_SOURCES} $<TARGET_OBJECTS:pierce_monocypher>)
target_include_directories(pierce_internal PUBLIC include src)
target_compile_definitions(pierce_internal PRIVATE PIERCE_BUILD)
target_compile_options(pierce_internal PRIVATE ${PIERCE_WARNINGS})
if(WIN32)
    target_link_libraries(pierce_internal PUBLIC ws2_32 iphlpapi bcrypt)
else()
    target_link_libraries(pierce_internal PUBLIC Threads::Threads)
endif()

if(PIERCE_BUILD_TESTS)
    enable_testing()
    file(GLOB PIERCE_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_*.c)
    foreach(t ${PIERCE_TESTS})
        get_filename_component(name ${t} NAME_WE)
        add_executable(${name} ${t})
        target_link_libraries(${name} PRIVATE pierce_internal)
        target_compile_options(${name} PRIVATE ${PIERCE_WARNINGS})
        add_test(NAME ${name} COMMAND ${name})
    endforeach()
    if(TARGET test_udp)
        target_sources(test_udp PRIVATE tools/relay_loop.c)
    endif()
endif()

if(PIERCE_BUILD_TOOLS AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tools/pierce_relay.c)
    add_executable(pierce-relay tools/pierce_relay.c tools/relay_loop.c)
    target_link_libraries(pierce-relay PRIVATE pierce_internal)
    target_compile_options(pierce-relay PRIVATE ${PIERCE_WARNINGS})
endif()

if(PIERCE_BUILD_EXAMPLES)
    file(GLOB PIERCE_EXAMPLES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/examples/*.c)
    foreach(e ${PIERCE_EXAMPLES})
        get_filename_component(name ${e} NAME_WE)
        add_executable(${name} ${e})
        target_link_libraries(${name} PRIVATE pierce)
        target_compile_options(${name} PRIVATE ${PIERCE_WARNINGS})
    endforeach()
endif()
